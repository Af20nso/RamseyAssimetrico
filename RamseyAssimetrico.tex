\section{Preliminares e Notação}
% ============================================================

Denotamos por $\Gnp$ o grafo aleatório binomial com $n$ vértices em que cada aresta
está presente independentemente com probabilidade $p = p(n)$.
Dizemos que $\Gnp$ satisfaz uma propriedade \emph{assintoticamente quase certamente}
(\whp) se a probabilidade de satisfazê-la tende a $1$ quando $n \to \infty$.

\begin{definition}[Limiar e tipos de limiar]
  Seja $\mathcal{P}$ uma propriedade monótona crescente de grafos.
  Uma função $\hat{p} = \hat{p}(n)$ é um \textbf{limiar grosseiro} (\emph{coarse threshold})
  para $\mathcal{P}$ em $\Gnp$ se
  \[
    \lim_{n\to\infty} \Pr[\Gnp \in \mathcal{P}] =
    \begin{cases}
      0 & \text{se } p = o(\hat{p}),      \\
      1 & \text{se } p = \omega(\hat{p}).
    \end{cases}
  \]
  Dizemos que $\hat{p}$ é um \textbf{limiar semi-agudo} (\emph{semi-sharp threshold})
  se existem constantes $c, C > 0$ tais que
  \[
    \lim_{n\to\infty} \Pr[\Gnp \in \mathcal{P}] =
    \begin{cases}
      0 & \text{se } p \le c\,\hat{p}, \\
      1 & \text{se } p \ge C\,\hat{p}.
    \end{cases}
  \]
  O \textbf{$0$-enunciado} refere-se à afirmação para $p$ abaixo do limiar,
  e o \textbf{$1$-enunciado} à afirmação para $p$ acima do limiar.
\end{definition}

\begin{definition}[Densidades de um grafo]
  \label{def:densidades}
  Dado um grafo $H$, definem-se:
  \begin{itemize}
    \item A \textbf{densidade} de $H$:
          $d(H) := \dfrac{e(H)}{v(H)}$, onde $e(H) = |E(H)|$ e $v(H) = |V(H)|$.

    \item A \textbf{densidade máxima} de $H$:
          \[
            \maxdens(H) := \max\!\left\{\frac{e(J)}{v(J)} : J \subseteq H,\; v(J) \ge 1\right\}.
          \]
          Dizemos que $H$ é \textbf{balanceado} se $\maxdens(H) = d(H)$, e
          \textbf{estritamente balanceado} se $\maxdens(H) > d(J)$ para todo
          $J \subsetneq H$ com $v(J) \ge 1$.

    \item A \textbf{2-densidade} de $H$ (para $v(H) \ge 3$):
          \[
            d_2(H) := \frac{e(H)-1}{v(H)-2}.
          \]

    \item A \textbf{2-densidade máxima} de $H$:
          \[
            \mdens(H) := \max\!\left\{d_2(J) : J \subseteq H,\; v(J) \ge 3\right\}
            \cup \left\{\tfrac{1}{2}\right\}.
          \]
          (A fração $\frac{1}{2}$ corresponde ao caso $H \cong K_2$.)
          Dizemos que $H$ é \textbf{$2$-balanceado} se $\mdens(H) = d_2(H)$, e
          \textbf{estritamente $2$-balanceado} se $\mdens(H) > d_2(J)$ para todo
          $J \subsetneq H$.
  \end{itemize}
\end{definition}

\begin{example}
  Para o clique $K_k$ com $k \ge 3$:
  $\mdens(K_k) = \dfrac{\binom{k}{2}-1}{k-2} = \dfrac{k+1}{2}$.
  Para o ciclo $C_\ell$ com $\ell \ge 3$:
  $\mdens(C_\ell) = \dfrac{\ell-1}{\ell-2}$.
  Em particular,
  $\mdens(K_5) = 3$, $\mdens(C_5) = \frac{4}{3}$, $\mdens(C_4) = \frac{3}{2}$.
\end{example}

\begin{definition}[$H$-fechamento]
  \label{def:H-fechado}
  Dado um grafo $G$ e um grafo $H$, dizemos que:
  \begin{itemize}
    \item Uma aresta $e \in E(G)$ é \textbf{$H$-fechada} se pertence a pelo menos duas cópias de~$H$ em~$G$.
    \item Uma cópia $H' \subseteq G$ de $H$ é \textbf{$H$-fechada} se pelo menos três de suas arestas são $H$-fechadas.
    \item O grafo $G$ é \textbf{$H$-fechado} se todo vértice e toda aresta de~$G$ pertencem a pelo menos uma cópia de~$H$, e toda cópia de~$H$ em~$G$ é $H$-fechada.
  \end{itemize}
\end{definition}

\begin{definition}[$H$-Bloco]
  \label{def:H-bloco}
  Um grafo $G$ é um \textbf{$H$-bloco} se $G$ é $H$-fechado e, para todo subconjunto próprio não-vazio de arestas $E' \subsetneq E(G)$, existe uma cópia $H'$ de $H$ em $G$ tal que $E(H') \cap E' \ne \emptyset$ e $E(H') \setminus E' \ne \emptyset$.
\end{definition}

\section{Propriedade de Ramsey em Grafos Aleatórios}
% ============================================================

\begin{definition}[Propriedade $r$-Ramsey]
  Para grafos $G$ e $H$ e um inteiro $r \ge 2$, escrevemos
  $G \ramsey{r} H$ para denotar a propriedade de que,
  em toda coloração das arestas de $G$ com $r$ cores,
  existe pelo menos uma cópia monocromática de $H$ em $G$.
\end{definition}

O seguinte resultado é o marco fundamental da teoria:

\begin{theorem}[Rödl--Ruciński \cite{rodl1993lower, rodl1994random, rodl1995threshold}]
  \label{thm:rodl-rucinski}
  Seja $r \ge 2$ um inteiro e seja $H$ um grafo que não é uma floresta de estrelas
  (e, no caso $r = 2$, também não é um caminho de comprimento~$3$).
  Existem constantes positivas $c = c(H,r)$ e $C = C(H,r)$ tais que
  \[
    \lim_{n\to\infty} \Pr\!\bigl[\Gnp \ramsey{r} H\bigr] =
    \begin{cases}
      0 & \text{se } p \le c\, n^{-1/\mdens(H)}, \\
      1 & \text{se } p \ge C\, n^{-1/\mdens(H)}.
    \end{cases}
  \]
  Em particular, $\hat{p}(n) = n^{-1/\mdens(H)}$ é o limiar semi-agudo para
  a propriedade $r$-Ramsey de $H$ em $\Gnp$.
\end{theorem}

Uma prova curta e elegante do Teorema~\ref{thm:rodl-rucinski} foi dada por
Nenadov e Steger \cite{nenadov2016short}, usando a técnica de
\emph{containers} de hipergrafos.

\section{Propriedade Anti-Ramsey em Grafos Aleatórios} \label{sec:antiramsey}
% ============================================================

\begin{definition}[Propriedade anti-Ramsey]
  \label{def:antiramsey}
  Dados grafos $G$ e $H$, escrevemos $G \rb H$ para denotar a
  \textbf{propriedade anti-Ramsey} de $G$ com respeito a $H$:
  para toda coloração própria das arestas de $G$
  (com número arbitrário de cores),
  existe pelo menos uma cópia \emph{arco-íris} de $H$ em $G$,
  isto é, uma cópia de $H$ em que cada aresta recebe uma cor distinta.
\end{definition}

\begin{remark}
  A propriedade $G \rb H$ é monótona crescente em $G$.
  Logo, para qualquer grafo fixo $H$, o grafo aleatório $\Gnp$ admite
  um limiar $p_H^{\mathrm{rb}} = p_H^{\mathrm{rb}}(n)$ para essa propriedade
  (cf.~Bollobás--Thomason).
\end{remark}

A questão central é determinar $p_H^{\mathrm{rb}}$ em termos de parâmetros
combinatórios de $H$.

\subsection{1-Enunciado: limite superior para o limiar}

O primeiro resultado geral foi obtido por Rödl e Tuza~\cite{rodl1992rainbow}
para ciclos, e depois estendido por Kohayakawa, Konstadinidis e Mota:

\begin{theorem}[Kohayakawa--Konstadinidis--Mota \cite{kohayakawa2014anti}]
  \label{thm:1statement}
  Seja $H$ um grafo qualquer. Existe uma constante $C > 0$ tal que,
  se $p = p(n) \ge C\, n^{-1/\mdens(H)}$, então \whp\
  $\Gnp \rb H$.
  Em particular, $p_H^{\mathrm{rb}} \le n^{-1/\mdens(H)}$.
\end{theorem}

Em outras palavras, o limiar anti-Ramsey nunca está acima do limiar Ramsey clássico.

\subsection{0-Enunciado: limite inferior — o \emph{framework} de Nenadov et al.}

Para o limite inferior, Nenadov, Person, Škorić e Steger desenvolveram um
arcabouço geral que reduz o problema probabilístico a uma questão determinística:

\begin{theorem}[Nenadov--Person--Škorić--Steger \cite{nenadov2017algorithmic}]
  \label{thm:framework}
  Seja $H$ um grafo estritamente $2$-balanceado.
  Se todo grafo $G$ tal que $G \rb H$ satisfaz $\maxdens(G) > \mdens(H)$,
  então existe uma constante $c > 0$ tal que, se $p \le c\, n^{-1/\mdens(H)}$,
  então \whp\ $\Gnp \not\rb H$.

  Em outras palavras, verificar a condição determinística acima
  é suficiente para obter o $0$-enunciado semi-agudo.
\end{theorem}

\begin{definition}[Condição determinística anti-Ramsey]
  \label{def:cond-det}
  Dado um grafo $H$, dizemos que $H$ \textbf{satisfaz a condição}~(D)
  se todo grafo $G$ tal que $G \rb H$ satisfaz $\maxdens(G) > \mdens(H)$.
\end{definition}

\subsection{Resultados para classes específicas de grafos}

\subsubsection*{Ciclos}

\begin{theorem}[Barros--Cavalar--Mota--Parczyk \cite{AntiCycles}]
  \label{thm:anticycles}
  O limiar $p_{C_\ell}^{\mathrm{rb}}$ para a propriedade anti-Ramsey do ciclo $C_\ell$
  satisfaz:
  \begin{itemize}
    \item Se $\ell \ge 5$, então
          $p_{C_\ell}^{\mathrm{rb}} = n^{-1/\mdens(C_\ell)} = n^{-(\ell-2)/(\ell-1)}$
          (limiar semi-agudo).
    \item Para $\ell = 4$:
          $p_{C_4}^{\mathrm{rb}} = n^{-3/4} \ll n^{-1/\mdens(C_4)} = n^{-2/3}$.
    \item Para $\ell = 3$: a condição de ser grafo arco-íris é satisfeita
          trivialmente, logo
          $p_{C_3}^{\mathrm{rb}} = n^{-1}$ (limiar para a presença de um triângulo).
  \end{itemize}
\end{theorem}

\subsubsection*{Grafos completos}

\begin{theorem}[Kohayakawa--Mota--Parczyk--Schnitzer \cite{AntiComplete}]
  \label{thm:anticomplete}
  Para o clique $K_k$, o limiar $p_{K_k}^{\mathrm{rb}}$ satisfaz:
  \begin{itemize}
    \item Se $k \ge 5$, então
          $p_{K_k}^{\mathrm{rb}} = n^{-1/\mdens(K_k)} = n^{-2/(k+1)}$
          (limiar semi-agudo).
    \item Para $k = 4$:
          $p_{K_4}^{\mathrm{rb}} = n^{-7/15} \ll n^{-1/\mdens(K_4)} = n^{-2/5}$.
    \item Para $k = 3$: trivialmente $p_{K_3}^{\mathrm{rb}} = n^{-1}$.
  \end{itemize}
\end{theorem}

\subsubsection*{Caso geral: 2-densidade alta}

\begin{theorem}[Kuperwasser \cite{kuperwasser2026anti}]
  \label{thm:kuperwasser}
  Seja $H$ um grafo com $\mdens(H) \ge 19$.
  Existem constantes $c_0 < c_1$ tais que
  \[
    \lim_{n\to\infty}\Pr\!\bigl[\Gnp \rb H\bigr] =
    \begin{cases}
      0 & \text{se } p \le c_0\, n^{-1/\mdens(H)}, \\
      1 & \text{se } p \ge c_1\, n^{-1/\mdens(H)}.
    \end{cases}
  \]
\end{theorem}

O Teorema~\ref{thm:kuperwasser} é provado verificando a condição determinística
do Teorema~\ref{thm:framework} através do seguinte resultado:

\begin{theorem}[Kuperwasser \cite{kuperwasser2026anti}]
  \label{thm:kuperwasser-det}
  Todo grafo $G$ com $\maxdens(G) \ge 18$ admite uma coloração própria das arestas
  tal que todo subgrafo arco-íris tem $2$-densidade estritamente menor que $\maxdens(G)$.
  Em particular, se $\mdens(H) > 18$, então $H$ satisfaz a condição~\emph{(D)}.
\end{theorem}

\subsection{A tabela de limiares conhecidos}

A tabela a seguir resume o estado atual dos limiares para a propriedade anti-Ramsey,

\subsection{Questão em aberto: grafos de cintura grande}

Os resultados acima revelam um padrão: o limiar anti-Ramsey diverge de
$n^{-1/\mdens(H)}$ exatamente para grafos ``pequenos'' de cintura baixa
($K_3$, $K_4$, $C_4$). Essa observação motiva a seguinte questão central
deste projeto de doutorado:

\begin{question}
  \label{q:cintura}
  Seja $H$ um grafo com cintura (comprimento do menor ciclo) estritamente maior que~$4$.
  É verdade que $p_H^{\mathrm{rb}} = n^{-1/\mdens(H)}$?
  Equivalentemente, $H$ sempre satisfaz a condição~\emph{(D)}
  (Definição~\ref{def:cond-det}) quando $\mathrm{girth}(H) > 4$?
\end{question}

\section{Propriedade Ramsey Restrita (Constrained Ramsey)} \label{sec:cram}
% ============================================================

Este problema, estudado em \cite{behague2025thresholds}, generaliza tanto
a propriedade de Ramsey clássica quanto a anti-Ramsey.

\begin{definition}[Propriedade Ramsey restrita]
  \label{def:cram}
  Dados grafos $H_1$ e $H_2$, escrevemos $G \cram (H_1, H_2)$ para denotar
  a \textbf{propriedade Ramsey restrita} (\emph{constrained Ramsey}):
  para toda coloração própria das arestas de $G$,
  existe uma cópia monocromática de $H_1$ \emph{ou} uma cópia arco-íris de $H_2$.
\end{definition}

\begin{remark}
  Quando $H_2$ é uma floresta, qualquer coloração própria tem trivialmente uma
  cópia arco-íris de $H_2$ (desde que o grafo seja grande o suficiente). Logo,
  o caso interessante é quando $H_2$ não é uma floresta.

  Quando $H_1 = $ estrela com $k$ arestas $K_{1,k}$ e $H_2 = H$, a propriedade
  $G \cram (K_{1,k}, H)$ equivale a dizer que toda coloração \emph{localmente
    $(k-1)$-limitada} de $G$ (em que cada cor aparece no máximo $k-1$ vezes por vértice)
  contém uma cópia arco-íris de $H$. Assim,
  a propriedade anti-Ramsey é o caso particular $k = 2$
  (colorações próprias = colorações localmente $1$-limitadas).
\end{remark}

\subsection{Resultados principais de Behague et al.}

\begin{theorem}[Behague--Hancock--Hyde--Letzter--Morrison \cite{behague2025thresholds}]
  \label{thm:behague-1-stmt}
  Sejam $k \ge 2$ e $H$ um grafo qualquer.
  Existe uma constante $C > 0$ tal que, se $p \ge C\, n^{-1/\mdens(H)}$, então
  \whp\ $\Gnp \cram (K_{1,k}, H)$.
\end{theorem}

O Teorema~\ref{thm:behague-1-stmt} generaliza o Teorema~\ref{thm:1statement}.

\begin{definition}[Densidades relevantes para Ramsey restrito]
  Para o problema restrito com $H_1 = K_{1,k}$, definem-se:
  \begin{align*}
    m_a(H) & := \inf\!\bigl\{\maxdens(G) : G \cram (K_{1,k}, H)\bigr\}                                                        \\
           & \phantom{:}= \text{densidade máxima do menor grafo} \ G\ \text{que força }                                       \\
           & \phantom{= \text{densidade máxima do menor grafo} G \text{que força}} \text{Ramsey restrito para } (K_{1,k}, H).
  \end{align*}
  O limiar candidato natural é então $n^{-1/m_a(H)}$.
\end{definition}

\begin{theorem}[Behague et al.\ \cite{behague2025thresholds} -- $0$-enunciado geral]
  \label{thm:behague-0-stmt}
  Sejam $k \ge 3$ e $H$ um grafo que não é floresta.
  \begin{enumerate}
    \item[(a)] Se $k \ge 4$, \emph{ou} $\mdens(H) \ne 2$, \emph{ou}
      $\mdens(H) = 2$ e $H$ contém um subgrafo estritamente $2$-balanceado
      $J \ne K_3$ com $\mdens(J) = 2$,
      então existem $c, C > 0$ tais que
      \[
        \lim_{n\to\infty} \Pr\!\bigl[\Gnp \cram (K_{1,k}, H)\bigr] =
        \begin{cases}
          0 & \text{se } p \le c\, n^{-1/\mdens(H)}, \\
          1 & \text{se } p \ge C\, n^{-1/\mdens(H)}.
        \end{cases}
      \]
    \item[(b)] Se $k = 3$ e $\mdens(H) = 2$ e o único subgrafo estritamente
      $2$-balanceado de $H$ com $2$-densidade~$2$ é $K_3$ (e.g.\ $H = K_3$),
      então o limiar é mais fraco: coarse threshold em $n^{-1/2}$.
  \end{enumerate}
\end{theorem}

\section{Propriedade Ramsey Canônica Assimétrica} \label{sec:canonico}
% ============================================================

A teoria canônica de Ramsey estuda estruturas que emergem em colorações
\emph{arbitrárias} (sem restrição de número de cores).

\begin{definition}[Coloração canônica]
  \label{def:canonica}
  Seja $c : E(G) \to \mathbb{N}$ uma coloração arbitrária das arestas de $G$,
  com uma ordenação total $v_1 < v_2 < \cdots < v_n$ dos vértices de $G$.
  Um subgrafo $H \subseteq G$ possui uma \textbf{coloração canônica} se
  satisfaz um dos seguintes padrões:
  \begin{enumerate}
    \item \textbf{Monocromático:} todas as arestas de $H$ têm a mesma cor.
    \item \textbf{Arco-íris:} todas as arestas de $H$ têm cores distintas.
    \item \textbf{Lexicográfico (min-lex):} a cor de $v_iv_j$ (com $v_i < v_j$)
          depende unicamente de $v_i$.
    \item \textbf{Lexicográfico (max-lex):} a cor de $v_iv_j$ (com $v_i < v_j$)
          depende unicamente de $v_j$.
  \end{enumerate}
\end{definition}

\begin{definition}[Propriedade Ramsey canônica]
  \label{def:prop-canonica}
  Dado um grafo ordenado $G$ (com uma ordenação total fixada nos vértices) e um grafo $H$,
  dizemos que $G$ satisfaz a \textbf{propriedade Ramsey canônica para $H$},
  e escrevemos $G \xrightarrow{\mathrm{can}} H$, se toda coloração arbitrária das arestas de $G$
  contém uma cópia canônica de $H$ (isto é, monocromática, arco-íris, ou lexicográfica min-lex).
\end{definition}

\begin{theorem}[Erdős--Rado \cite{erdos1950combinatorial}]
  Para qualquer inteiro $\ell \ge 1$, existe $N = N(\ell)$ tal que toda coloração
  arbitrária das arestas de $K_N$ contém uma cópia canônica de $K_\ell$.
\end{theorem}

O resultado de Erdős--Rado foi trazido para o contexto aleatório por Kamčev e Schacht:

\begin{theorem}[Kamčev--Schacht \cite{kamcev2023canonical}]
  \label{thm:kamcev}
  Para todo $\ell \ge 4$, o limiar para a propriedade de que toda coloração
  arbitrária de $\Gnp$ contenha uma cópia canônica de $K_\ell$ é
  $\hat{p}(n) = n^{-2/(\ell+1)} = n^{-1/\mdens(K_\ell)}$.
\end{theorem}

\subsection{Cenário assimétrico}

\begin{definition}[Propriedade Ramsey canônica assimétrica]
  \label{def:assimetrico}
  Seja $G$ um grafo com uma ordenação total fixada nos vértices.
  Dados grafos $H_1$, $H_2$ e $H_3$, escrevemos
  \[
    G \xrightarrow{\mathrm{can}} (H_1, H_2, H_3)
  \]
  para denotar a seguinte propriedade: para toda coloração arbitrária das
  arestas de $G$, existe pelo menos
  \begin{itemize}
    \item uma cópia \textbf{monocromática} de $H_1$, ou
    \item uma cópia \textbf{arco-íris} de $H_2$, ou
    \item uma cópia \textbf{lexicográfica min-lex} de $H_3$.
  \end{itemize}
  (Adotamos ao longo deste trabalho a convenção de que por ``cópia lexicográfica'' entendemos sempre a cópia min-lex.)
\end{definition}

\begin{question}
  \label{q:assimetrico}
  Qual é o comportamento da função limiar $\hat{p}(n)$ para a propriedade
  $\Gnp \xrightarrow{\mathrm{can}} (H_1, H_2, H_3)$
  quando $H_1$, $H_2$ e $H_3$ não são todos iguais?
  Em particular, como o limiar depende das densidades $\mdens(H_i)$?
\end{question}

% ============================================================
\section{O Caso Assimétrico \texorpdfstring{$(K_3, K_4, K_5)$}{(K3, K4, K5)}}
% ============================================================

Queremos determinar o limiar exato para a propriedade $\Gnp \xrightarrow{\mathrm{can}} (K_3,K_4,K_5)$.
Note que, para $p \ll n^{-1/\mdens(K_3)} = n^{-1/2}$, o grafo aleatório \whp\ não possui triângulos, tornando a propriedade trivialmente falsa (através de uma coloração arco-íris, por exemplo).
Por outro lado, para $p \gg n^{-1/\mdens(K_5)} = n^{-1/3}$, a existência de um $K_5$ canônico é garantida assintoticamente quase certamente (a.q.c.), o que força a existência de uma das estruturas proibidas.

O resultado principal que buscamos formalizar é o seguinte:

\begin{theorem}
  \label{thm:main-can}
  Existe um limiar semi-agudo para a propriedade $\Gnp \xrightarrow{\mathrm{can}} (K_3,K_4,K_5)$ em $\hat{p}(n) = n^{-1/3}$. Ou seja, existem constantes $c, C > 0$ tais que:
  \[
    \lim_{n \to \infty} \mathbb{P}(\Gnp \xrightarrow{\mathrm{can}} (K_3,K_4,K_5)) =
    \begin{cases}
      1 & \text{se } p \ge C n^{-1/3}, \\
      0 & \text{se } p \le c n^{-1/3}.
    \end{cases}
  \]
\end{theorem}

Para a prova do Teorema~\ref{thm:main-can}, utilizamos os conceitos de $H$-fechamento (Definição~\ref{def:H-fechado}) e $H$-bloco (Definição~\ref{def:H-bloco}). A ponte probabilística fundamental que conecta o limite macroscópico do grafo aleatório $\Gnp$ aos blocos de tamanho pequeno é dada pelo seguinte resultado de Nenadov et al.~\cite{nenadov2017algorithmic}, cuja prova é baseada no método de recipientes (containers) hipergráficos e foi aplicada a problemas de Ramsey aleatórios por Nenadov e Steger~\cite{nenadov2016short}:

\begin{theorem}[\cite{nenadov2017algorithmic}, Lemma 3.1]
  \label{thm:nenadov-steger}
  Sejam $H$ um grafo estritamente $2$-balanceado e $c > 0$ uma constante suficientemente pequena. Se $p \le c n^{-1/m_2(H)}$, então com alta probabilidade toda componente conexa do $H$-fechamento de $\Gnp$ tem ordem limitada por uma constante $C_0 = C_0(c)$. Em particular, cada $H$-bloco de $\Gnp$ tem tamanho $O(1)$ \whp.
\end{theorem}

Apresentamos a demonstração completa do Teorema~\ref{thm:main-can} com base nessa estrutura:

\begin{proof}[Demonstração do Teorema~\ref{thm:main-can}]
  Dividimos a prova em duas partes, demonstrando primeiro o enunciado de $1$ (limite superior) e em seguida o de $0$ (limite inferior).

  \medskip
  \noindent\textbf{1. O $1$-enunciado ($p \ge C n^{-1/3}$):}
  Queremos provar que, para uma constante $C$ suficientemente grande, o grafo aleatório $G \sim \Gnp$ satisfaz a propriedade Ramsey assimétrica $G \xrightarrow{\mathrm{can}} (K_3, K_4, K_5)$ com alta probabilidade.
  Para tanto, estabelecemos a seguinte redução de dominância matemática: a validade da propriedade Ramsey canônica simétrica $G \xrightarrow{\mathrm{can}} K_5$ implica diretamente a validade da propriedade assimétrica $G \xrightarrow{\mathrm{can}} (K_3, K_4, K_5)$.
  De fato, seja $\chi: E(G) \to \mathbb{N}$ uma coloração arbitrária das arestas de $G$. Se $G \xrightarrow{\mathrm{can}} K_5$, então existe por definição uma cópia $H \subseteq G$ de $K_5$ ($H \cong K_5$) que é canônica sob $\chi$. Analisamos os três cenários possíveis para esta cópia canônica $H$:
  \begin{itemize}
    \item \textbf{Cenário 1 ($H$ é monocromática):} Todas as $10$ arestas de $H$ possuem a mesma cor sob $\chi$. Neste caso, qualquer triângulo $K_3$ contido em $H$ possui suas $3$ arestas coloridas com a mesma cor, configurando uma cópia monocromática de $K_3$ em $G$.
    \item \textbf{Cenário 2 ($H$ é arco-íris):} As $10$ arestas de $H$ possuem cores duas a duas distintas sob $\chi$. Consequentemente, qualquer sub-clique $K_4 \subseteq H$ possui suas $6$ arestas coloridas com cores duas a duas distintas, configurando uma cópia arco-íris de $K_4$ em $G$.
    \item \textbf{Cenário 3 ($H$ é lexicográfica min-lex):} Por definição de cópia canônica de $K_5$ (uma vez excluída a cópia max-lex por convenção de simetria de ordenação total), o grafo $G$ contém uma cópia lexicográfica min-lex de $K_5$ sob $\chi$.
  \end{itemize}
  Como a existência de um $K_5$ canônico sob $\chi$ implica necessariamente a presença de um $K_3$ monocromático, de um $K_4$ arco-íris ou de um $K_5$ lexicográfico min-lex, a implicação $(G \xrightarrow{\mathrm{can}} K_5) \implies (G \xrightarrow{\mathrm{can}} (K_3, K_4, K_5))$ está demonstrada.
  Sob o Teorema de Kamčev--Schacht~\cite{kamcev2023canonical} (Teorema~\ref{thm:kamcev} com $\ell = 5$), o limiar para a propriedade canônica simétrica $\Gnp \xrightarrow{\mathrm{can}} K_5$ ocorre exatamente em $\hat{p}(n) = n^{-1/\mdens(K_5)} = n^{-1/3}$.
  Desta forma, para $p \ge C n^{-1/3}$ com $C$ suficientemente grande, temos $\Gnp \xrightarrow{\mathrm{can}} K_5$ \whp, o que garante a validade da propriedade Ramsey assimétrica $\Gnp \xrightarrow{\mathrm{can}} (K_3, K_4, K_5)$ \whp, completando a prova do $1$-enunciado.

  \medskip
  \noindent\textbf{2. O $0$-enunciado ($p \le c n^{-1/3}$):}
  Queremos construir \whp\ uma coloração válida de $G \sim \Gnp$ que evite simultaneamente as três configurações proibidas.
  Visto que $p \le c n^{-1/3} = c n^{-1/\mdens(K_5)}$, aplicamos o Teorema~\ref{thm:nenadov-steger} para o grafo estritamente $2$-balanceado $H = K_5$ (onde $\mdens(K_5) = 3$). Logo, com alta probabilidade, toda componente conexa de seu $K_5$-fechamento (e portanto cada $K_5$-bloco $B_j$ contido em $G$) possui tamanho $O(1)$.
  Além disso, a densidade máxima de cada bloco satisfaz $\maxdens(B_j) < 3$ \whp\ (pois o surgimento de subgrafos densos de tamanho pequeno com $\maxdens \ge 3$ sob o regime $p \le c n^{-1/3}$ possui probabilidade tendendo a zero para $c > 0$ suficientemente pequeno).
  A coloração das arestas de $G$ é definida de forma determinística utilizando o algoritmo \textsc{ColoraçãoCan}:

  \begin{algorithm}[H]
    \caption{\textsc{ColoraçãoCan}$(G = (V, E),\; \sigma)$}
    \label{alg:coloracao-can}
    \begin{algorithmic}[1]
      \State $\hat{G} \leftarrow G$
      \State $\mathrm{col} \leftarrow 0$
      \While{$\exists\, e_1, e_2 \in E(\hat{G})$ : $e_1 \equiv_{K_5} e_2$ em $\hat{G}$ e $e_1 \cap e_2 = \emptyset$}
      \State colorir $e_1, e_2$ com col
      \State $\hat{G} \leftarrow \hat{G} \setminus \{e_1, e_2\}$ e $\mathrm{col} \leftarrow \mathrm{col} + 1$
      \EndWhile
      \For{cada $uv \in E(\hat{G})$}
      \State $\chi(uv) \leftarrow \varphi\!\bigl(\min_\sigma\{u,v\}\bigr)$
      \EndFor
      \While{$\exists\, e \in E(\hat{G})$ : $e$ não pertence a nenhuma cópia de $K_5$ em $\hat{G}$}
      \State $\hat{G} \leftarrow \hat{G} \setminus \{e\}$
      \EndWhile
      \State Remover vértices isolados de $\hat{G}$
      \State $\{B_1, \ldots, B_k\} \leftarrow K_5\text{-blocos}$ obtidos de $\hat{G}$
      \For{cada bloco $B_j$}
      \State colorir $B_j$ sem $K_5$ lexicográfico, $K_3$ monocromático e $K_4$ arco-íris,
      \Statex \hspace{2.85em} usando conjunto de cores distinto dos demais blocos
      \EndFor
    \end{algorithmic}
  \end{algorithm}

  O algoritmo particiona as arestas de $G$ em três classes principais: as arestas que não pertencem a nenhuma cópia de $K_5$ (que recebem cores únicas e exclusivas no passo 8), as arestas equivalentes desmembradas no passo 3, e as arestas restantes nos blocos isolados de tamanho pequeno $B_j$.
  Toda cópia de $K_5$ em $G$ está contida inteiramente em algum $K_5$-bloco $B_j$. Como colorimos cada bloco $B_j$ com conjuntos de cores disjuntos dos demais e de qualquer outra aresta fora de $B_j$, qualquer potencial cópia de $K_3$ monocromático, $K_4$ arco-íris ou $K_5$ lexicográfico no grafo global deve estar inteiramente contida dentro de um único bloco $B_j$.
  Porém, como cada bloco $B_j$ possui tamanho $O(1)$ e satisfaz $\maxdens(B_j) < 3$ \whp, o Lema~\ref{lem:0-statement-K5} garante que cada $B_j$ admite uma coloração válida livre de tais estruturas proibidas. As demais arestas de floresta e conexões externas recebem cores únicas exclusivas, impossibilitando a formação de novos triângulos monocromáticos ou padrões lexicográficos.
  Assim, a coloração resultante é válida livre de $K_3$ monocromático, $K_4$ arco-íris e $K_5$ lexicográfico com alta probabilidade, o que prova o $0$-enunciado e conclui o teorema.
\end{proof}

\begin{definition}[Função de penalidade]
  \label{def:penalidade}
  Dado um grafo $G$, definimos a sua penalidade como:
  \[
    p(G) \;:=\; e(G) - 3\,v(G) + 5.
  \]
\end{definition}

A função $p$ é normalizada de forma que o $K_5$ inicial tenha penalidade nula:
\[
  p(K_5) = 10 - 3 \cdot 5 + 5 = 0.
\]
A condição de densidade máxima $\maxdens(B) < 3$ garante que $e(G) \le 3\,v(G) - 1$ para qualquer subgrafo $G \subseteq B$, o que implica $p(G) \le 4$. Esta limitação superior uniforme $p(B) \le 4$ é crucial para controlar a complexidade estrutural do bloco.

Seguindo as estratégias de Nenadov et al.~\cite{nenadov2017algorithmic} e~\cite{AntiComplete}, a construção indutiva de subgrafos de $B$ a partir de um subgrafo base $H_0 \cong K_5$ (ou $K_6$) é classificada em quatro tipos estruturais de expansão:

\begin{itemize}
  \item \textbf{$R_\ell$ (Expansão regular):} Adicionam-se $3$ novos vértices e $9$ novas arestas que, juntamente com uma aresta existente, formam um novo $K_5$. Neste caso, a variação da penalidade é $\Delta p = 9 - 3(3) = 0$.
  \item \textbf{$T_\ell$ (Expansão sobre triângulo):} Adicionam-se $2$ novos vértices e $7$ novas arestas que, apoiadas em um triângulo preexistente, completam um novo $K_5$. Aqui, $\Delta p = 7 - 3(2) = 1$.
  \item \textbf{$Q_\ell$ (Expansão sobre $K_4$):} Adiciona-se $1$ novo vértice ligado a um $K_4$ existente através de $4$ novas arestas, completando um $K_5$. O incremento é $\Delta p = 4 - 3(1) = 1$.
  \item \textbf{$U_\ell$ (União sobre $K_5^-$):} Não se adicionam vértices; apenas $1$ nova aresta é inserida conectando dois vértices de um $K_5$ quase completo ($K_5 \setminus \{e\}$) já presente na estrutura. A variação é $\Delta p = 1 - 3(0) = 1$.
\end{itemize}

\begin{remark}
  \textbf{Compatibilidade com a definição de Bloco:} Como toda cópia de $K_5$ em um $K_5$-bloco $B$ deve ser $K_5$-fechada (ou seja, ter pelo menos $3$ arestas $K_5$-fechadas que pertencem a outras cópias de $K_5$ no bloco), a expansão regular $R_\ell$ (que compartilha apenas uma única aresta com a estrutura existente) não pode ocorrer isoladamente na construção interna de um $K_5$-bloco $B$ fechado. Ela é utilizada apenas para conectar blocos diferentes ou partes florestais em árvore do grafo global $G$. Portanto, a construção indutiva de um $K_5$-bloco $B$ consiste exclusivamente de passos degenerados $T_\ell$, $Q_\ell$ e $U_\ell$.
\end{remark}

As expansões degeneradas ($T_\ell$, $Q_\ell$ e $U_\ell$) aumentam a penalidade estritamente em $1$. Como iniciamos com $p(H_0) = 0$ e $p(B) \le 4$, concluímos que o número total destas expansões na construção do bloco $B$ é de no máximo $4$.

\begin{lemma}
  \label{lem:0-statement-K5}
  Seja $B$ um $K_5$-bloco com $\maxdens(B) < 3$. Então existe uma coloração $c: E(B) \to \mathbb{N}$ tal que:
  \begin{enumerate}
    \item[(i)] $c$ não possui $K_3$ monocromático;
    \item[(ii)] $c$ não possui $K_4$ arco-íris;
    \item[(iii)] $c$ não possui $K_5$ lexicográfico.
  \end{enumerate}
\end{lemma}

A prova do Lema~\ref{lem:0-statement-K5} utiliza as seguintes proposições e afirmações auxiliares:

\begin{proposition}[Equilíbrio de cores]
  \label{prop:equilibrio}
  Seja $B$ um $K_5$-bloco com $\maxdens(B) < 3$ e $\chi'$ a coloração auxiliar construída indutivamente sobre as expansões de $B$. Então é possível definir $\chi'$ de modo que, para toda cópia $H \cong K_5$ em $B$ e todo vértice $w \in V(H)$, $w$ possui ao menos um vizinho com $\chi'(e) = \text{claro}$ e ao menos um vizinho com $\chi'(e) = \text{escuro}$ dentro de $H$.
\end{proposition}

\begin{afirmacao}[Expansões de vértice preservam o equilíbrio]
  \label{afirm:vertex-exp}
  A propriedade da Proposição~\ref{prop:equilibrio} é preservada em cada passo dos tipos $R_\ell$, $T_\ell$ e $Q_\ell$.
\end{afirmacao}

\begin{afirmacao}[Limitação de $U_\ell$]
  \label{afirm:ulimit}
  Na construção de qualquer $K_5$-bloco $B$ com $\maxdens(B) < 3$, o número de expansões do tipo $U_\ell$ é no máximo $2$.
\end{afirmacao}

\begin{afirmacao}[$\chi''$ é uma coloração válida]
  \label{afirm:chipp}
  Dadas $\chi$ e $\chi'$ como acima, a coloração $\chi''(e) := (\chi(e), \chi'(e))$ evita $K_3$ monocromático, $K_4$ arco-íris e $K_5$ lexicográfico.
\end{afirmacao}

\begin{proof}[Prova do Lema~\ref{lem:0-statement-K5}]
  A coloração $\chi'' = (\chi, \chi')$ é construída combinando a coloração lexicográfica min-lex inicial $\chi$ (obtida sob uma ordenação total fixada $\sigma$) com a coloração auxiliar de duas cores $\chi'$ cuja existência e propriedades de equilíbrio são garantidas pela Proposição~\ref{prop:equilibrio}. A validade de $\chi''$ para evitar $K_3$ monocromático, $K_4$ arco-íris e $K_5$ lexicográfico é estabelecida diretamente pela Afirmação~\ref{afirm:chipp}, o que conclui a prova do Lema~\ref{lem:0-statement-K5}.
\end{proof}

\begin{proof}[Prova da Proposição~\ref{prop:equilibrio}]
  Pela Afirmação~\ref{afirm:vertex-exp}, a propriedade de equilíbrio é mantida a cada expansão dos tipos $R_\ell$, $T_\ell$ e $Q_\ell$. Pela Afirmação~\ref{afirm:ulimit}, o número de passos $U_\ell$ é no máximo $2$. Como cada passo $U_\ell$ apenas adiciona uma aresta a um $K_5^-$ existente, ele preserva as conexões e a coloração auxiliar já estabelecida para as demais arestas de cada vértice dentro daquela cópia de $K_5$. Portanto, a propriedade de equilíbrio é garantida para todas as cópias após todos os passos de construção, completando a prova da Proposição~\ref{prop:equilibrio}.
\end{proof}

\begin{proof}[Prova da Afirmação~\ref{afirm:chipp} ($\chi''$ é uma coloração válida)]
  Assumimos que $\chi'$ satisfaz a propriedade de equilíbrio da Proposição~\ref{prop:equilibrio} e verificamos cada uma das três propriedades de validade de $\chi''$:

  \medskip
  \noindent\textbf{1. Ausência de $K_5$ lexicográfico:}
  Sob $\chi$, toda cópia de $K_5$ em $B$ é lexicográfica min-lex. Isto significa que existe uma ordenação $\sigma$ e uma atribuição de cores de vértices $\varphi$ tais que para todo par $uv$, $\chi(uv) = \varphi(\min_\sigma\{u,v\})$.
  Abaixo, ilustramos este padrão lexicográfico min-lex inicial $\chi$ em uma cópia base de $K_5$:

  \begin{center}
    \begin{tikzpicture}[scale=1.3,
        vx/.style={circle,draw,fill=white,minimum size=16pt,font=\small}]
      % K5 vertices as regular pentagon
      \node[vx] (1) at (90:1.5)  {$v_1$};
      \node[vx] (2) at (18:1.5)  {$v_2$};
      \node[vx] (3) at (-54:1.5) {$v_3$};
      \node[vx] (4) at (-126:1.5){$v_4$};
      \node[vx] (5) at (162:1.5) {$v_5$};

      \draw[line width=1.8pt, red]    (1)--(3);
      \draw[line width=1.8pt, red]    (1)--(4);
      \draw[line width=1.8pt, red]    (1)--(5);
      % color b = phi(v2): edges 23,24,25

      \draw[line width=1.8pt, blue]   (2)--(4);
      \draw[line width=1.8pt, blue]   (2)--(5);

      % color d = phi(v4): edge 45
      \draw[line width=1.8pt, orange] (4)--(5);
    \end{tikzpicture}
  \end{center}
  \noindent
  No diagrama acima, mostramos a coloração min-lex $\chi$ de um $K_5$: as arestas incidentes ao vértice mínimo $v_1$ têm todas a mesma cor vermelha; as arestas incidentes a $v_2$ que não vão para $v_1$ têm cor azul; as arestas incidentes a $v_3$ têm cor verde (teal); e a aresta $v_4v_5$ tem cor laranja.

  Para que este $K_5$ permanecesse min-lex sob o produto cartesiano $\chi'' = (\chi, \chi')$, seria necessário que todas as arestas de cor $\chi(uv)$ incidente ao vértice mínimo $v_1$ recebessem também a mesma cor auxiliar sob $\chi'$. Ou seja, deveríamos ter $\chi'(v_1v_2) = \chi'(v_1v_3) = \chi'(v_1v_4) = \chi'(v_1v_5) = \text{claro}$ (ou todos escuro), o que significa que o vértice $v_1$ teria grau $4$ na cor claro e grau $0$ na cor escuro dentro deste $K_5$.
  No entanto, a Proposição~\ref{prop:equilibrio} garante que todo vértice (incluindo $v_1$) possui pelo menos um vizinho claro e pelo menos um vizinho escuro sob $\chi'$ dentro deste $K_5$. Portanto, nenhuma cópia de $K_5$ pode ser lexicográfica sob $\chi''$.

  \medskip
  \noindent\textbf{2. Ausência de $K_4$ arco-íris:}
  Suponha por contradição que existe uma cópia $K_4^* \subseteq B$ que é arco-íris sob $\chi''$.
  Isso significa que as $6$ arestas de $K_4^*$ possuem cores distintas sob o produto cartesiano $\chi'' = (\chi, \chi')$.
  Em particular, as $6$ arestas de $K_4^*$ devem possuir cores distintas sob a projeção $\chi$.
  Mas isso significa que $K_4^*$ é arco-íris sob a coloração inicial $\chi$.
  Contudo, a coloração $\chi$ é construída de modo a ser livre de $K_4$ arco-íris por definição (já que $\chi$ é uma coloração min-lex sobre uma ordenação linear, e é um fato conhecido que colorações lexicográficas min-lex não contêm $K_4$ arco-íris).
  Isso contradiz o pressuposto, logo $\chi''$ não contém $K_4$ arco-íris.

  \medskip
  \noindent\textbf{3. Ausência de $K_3$ monocromático:}
  Suponha que existe um triângulo monocromático $T \subseteq B$ sob $\chi''$.
  Então as $3$ arestas de $T$ devem ter a mesma cor em ambas as componentes:
  $\chi(xy) = \chi(yz) = \chi(xz) = c_0$ e $\chi'(xy) = \chi'(yz) = \chi'(xz) = c_1$.
  Visto que $T$ é monocromático sob $\chi$ e $\chi$ é min-lex sobre a ordenação $\sigma$, os vértices de $T$ ordenados como $x < y < z$ devem satisfazer $\chi(xy) = \chi(xz) = \varphi(x) = c_0$.
  Para que a terceira aresta $yz$ também tenha cor $c_0$, deveríamos ter $\chi(yz) = \varphi(y) = c_0$, o que implica $\varphi(x) = \varphi(y)$ (o que viola o fato de que a coloração é construída de modo que $\varphi$ atribui cores distintas a vértices distintos em qualquer bloco $B$ pequeno de tamanho $O(1)$, pois a função $\varphi$ é injetiva sobre a ordenação $\sigma$ do bloco).
  Assim, $\chi''(xy) \ne \chi''(yz)$, e o triângulo não pode ser monocromático sob $\chi''$, concluindo a demonstração.
\end{proof}

\begin{proof}[Prova da Afirmação~\ref{afirm:vertex-exp} (Expansões de vértice)]
  A prova é por indução no número de passos de expansão na construção do bloco $B$.
  Mantemos o seguinte invariante: após cada passo, toda cópia de $K_5$ presente no grafo
  atual admite uma decomposição $E(K_5) = E(C_5^{\text{claro}}) \cup E(C_5^{\text{escuro}})$
  (com todo vértice de grau $2$ em cada $C_5$), e todo triângulo presente tem exatamente
  $2$ arestas de uma cor e $1$ da outra (distribuição $2$-$1$).

  \medskip
  \noindent\textbf{Caso Base: $H_0 \cong K_5$.}
  O caso base $H_0 \cong K_5$ com vértices $v_1 < v_2 < v_3 < v_4 < v_5$ satisfaz o invariante pela construção explícita da coloração auxiliar $\chi': E(H_0) \to \{\text{claro}, \text{escuro}\}$: definimos as cores identificando as $10$ arestas de $K_5$ com as $10$ arestas de duas cópias de $C_5$: o \textbf{pentágono externo} (as arestas $v_1v_2, v_2v_3, v_3v_4, v_4v_5, v_5v_1$) pintado na cor vermelha (cor claro) e o \textbf{pentágono interno/estrela} (as diagonais $v_1v_3, v_3v_5, v_5v_2, v_2v_4, v_4v_1$) pintado na cor azul (cor escuro).

  \begin{center}
    \begin{tikzpicture}[scale=1.3,
        vx/.style={circle,draw,fill=white,minimum size=16pt,font=\small}]
      \node[vx] (1) at (90:1.5)  {$v_1$};

\begin{proof}[Prova da Afirmação~\ref{afirm:ulimit} (Limitação de $U_\ell$)]
  Dividimos a prova em dois casos, conforme o bloco $B$ contenha ou não uma cópia de $K_6$.

  \medskip
  \noindent\textit{Caso 2: O bloco $B$ contém uma cópia de $K_6$.}

  \medskip

  Como a $2$-densidade máxima do bloco satisfaz $\maxdens(B) < 3$, o bloco $B$ não pode conter nenhuma cópia de $K_7$ como subgrafo (uma vez que $m(K_7) = 21/7 = 3$). Assim, a maior cópia de clique completo que $B$ pode conter é $K_6$.

  Fixamos $H_0 \cong K_6$ como grafo inicial da construção indutiva de $B$. Como $B$ é um $K_5$-bloco e contém $K_6$, é possível iniciar a construção indutiva a partir deste subgrafo $K_6$ maximal e anexar os demais vértices por expansões sucessivas, seguindo a estrutura padrão descrita em~\cite{nenadov2017algorithmic} e \cite{AntiComplete}.

  \medskip
  \noindent\textit{Penalidade inicial.}
  Calculamos diretamente a penalidade de $H_0 \cong K_6$:
  \[
    p(K_6) \;=\; e(K_6) - 3\,v(K_6) + 5 \;=\; 15 - 3 \cdot 6 + 5 \;=\; 2.
  \]
  Como a densidade máxima do bloco satisfaz $\maxdens(B) < 3$, temos $e(B) \le 3\,v(B) - 1$, o que garante que $p(B) \le 4$. Logo, o orçamento total para passos degenerados a partir de $H_0$ é de no máximo $p(B) - p(K_6) \le 4 - 2 = 2$.

  \medskip
  \noindent\textit{Coloração auxiliar de $K_6$.}
  Rotulamos os vértices de $K_6$ como $1, 2, 3, 4, 5, 6$ e definimos $\chi'$ por:
  \begin{itemize}
    \item $C_6$ vermelho: $1\text{-}2\text{-}3\text{-}4\text{-}5\text{-}6\text{-}1$
          (as $6$ arestas do ciclo externo);
    \item $C_6$ azul: $1\text{-}3\text{-}5\text{-}1$ e $2\text{-}4\text{-}6\text{-}2$
          reunidos, ou seja, as $6$ arestas $13, 35, 15, 24, 46, 26$
          (o ciclo ``saltando de dois em dois'');
    \item $M_3$ azul: o emparelhamento perfeito $\{14, 25, 36\}$.
  \end{itemize}
  Tem-se $6 + 6 + 3 = 15 = \binom{6}{2}$ arestas, cobrindo exatamente $E(K_6)$.

  \medskip

      \draw[az] (v\a) -- (v\b);

  \medskip
  \noindent\textit{Verificação das propriedades invariantes.}

  A coloração acima satisfaz o invariante estabelecido na prova da
  Afirmação~\ref{afirm:vertex-exp}:

  \begin{itemize}
    \item Todo vértice de $K_6$ tem grau $2$ no $C_6$ vermelho, grau $2$ no $C_6$ azul
          e grau $1$ no $M_3$ vermelho, portanto grau total $3$ em arestas vermelhas e
          grau $2$ em arestas azuis.

    \item Qualquer cópia de $K_5$ em $K_6$ é obtida removendo um vértice $w$.
          Dentro desse $K_5$, cada vértice perde exatamente $1$ vizinho (o próprio $w$),
          mas mantém ao menos $1$ vizinho vermelho e $1$ vizinho azul, preservando o
          equilíbrio da Proposição~\ref{prop:equilibrio}.

    \item Todo triângulo de $K_6$ possui exatamente $2$ arestas vermelhas e $1$ azul,
          ou $1$ vermelha e $2$ azuis (pode-se verificar por inspeção direta nos
          $\binom{6}{3} = 20$ triângulos). Em particular, a distribuição $2$-$1$ vale,
          garantindo a aplicabilidade da Afirmação~\ref{afirm:vertex-exp}.
  \end{itemize}

  \medskip
  \noindent\textit{Como podem surgir passos $U_\ell$ a partir de $K_6$.}

  Um passo $U_\ell$ requer a presença de um $K_5^-$ no grafo atual, isto é, cinco
  vértices faltando exatamente uma aresta para formar $K_5$.
  No entanto, dentro de $K_6$ puro, qualquer conjunto de $5$ vértices induz um $K_5$
  completo; logo \emph{não existe} $K_5^-$ dentro de $H_0 = K_6$.

  Portanto, um passo $U_\ell$ nunca pode ser o primeiro passo após $H_0$: é
  necessário realizar ao menos um passo de expansão degenerado do tipo $T_\ell$ ou
  $Q_\ell$ antes, de modo a introduzir novos vértices e criar estruturas contendo um $K_5^-$.

  Cada um desses passos de  \medskip
  \noindent\textit{Caso 1: O bloco $B$ não contém nenhuma cópia de $K_6$.}

  \medskip

  Iniciamos a construção a partir de uma única cópia base $H_0 \cong K_5$, com
  $p(H_0) = 0$. O orçamento total para passos degenerados é $p(B) \le 4$.
  Seja $k$ o número de passos $Q_\ell$ aplicados diretamente sobre $H_0$ (ou seja,
  o número de vértices acoplados a $H_0$ via $Q_\ell$). Cada passo $Q_\ell$ consome $1$ unidade
  de orçamento. Analisamos todos os cenários para $k \ge 0$.

  \medskip
  \noindent\textbf{Subcaso 1a: $k \ge 2$.}
  Se $k \ge 2$ passos $Q_\ell$ são realizados diretamente sobre $H_0$, o orçamento de penalidade
  consumido é de pelo menos $2$. Como o orçamento total é de no máximo $4$, o orçamento
  restante para todo o resto da construção (incluindo todos os passos $U_\ell$) é de no máximo
  $4 - 2 = 2$. Visto que cada passo $U_\ell$ consome exatamente $1$ unidade de penalidade,
  o número total de passos $U_\ell$ é trivialmente limitado por $2$.

  \medskip
  \noindent\textbf{Subcaso 1b: $k = 1$.}
  Suponha que realizamos exatamente um único passo $Q_\ell$ sobre $H_0$. Este passo adiciona
  um vértice $u$ conectado a um $K_4 = \{a,b,c,d\} \subset H_0$, consumindo $1$ unidade de orçamento.
  O vértice restante de $H_0$ é denotado por $v$.

  Neste ponto, o orçamento restante é $3$. No entanto, mostramos que um passo $U_\ell$ é
  \emph{impossível} imediatamente após este $Q_\ell$: a única não-aresta cuja adição poderia
  fechar um $K_5^-$ é $\{u,v\}$, mas adicioná-la completaria um $K_6$ sobre o conjunto
  de vértices $\{u,v,a,b,c,d\}$ (pois $u$ e $v$ são ambos conectados a todo o $K_4 = \{a,b,c,d\}$).
  Como $B$ não contém $K_6$, nenhum passo $U_\ell$ pode ser o próximo passo da construção.

  Desta forma, deve-se realizar pelo menos mais um passo de expansão antes do primeiro $U_\ell$.
  Como expansões regulares do tipo $R_\ell$ não criam novos $K_5^-$ (pois compartilham apenas uma
  aresta, o que é insuficiente para formar o acoplamento de $K_3$ exigido por um $K_5^-$), este passo
  deve ser degenerado:
  \begin{itemize}
    \item Se for outro passo $Q_\ell$, caímos no cenário $k \ge 2$, onde o orçamento restante é $\le 2$.
    \item Se for um passo $T_\ell$, este consome $1$ unidade de orçamento. O orçamento restante para o
          resto da construção cai para no máximo $3 - 1 = 2$, limitando o número de passos $U_\ell$ a no máximo $2$.
  \end{itemize}
  Portanto, em todos os caminhos de construção com $k = 1$, o número de passos $U_\ell$ é no máximo $2$.

  \medskip
  \noindent\textbf{Subcaso 1c: $k = 0$.}
  Neste caso, nenhum passo $Q_\ell$ é realizado diretamente sobre $H_0$. O orçamento total disponível é $4$.
  Para que qualquer passo $U_\ell$ seja possível, o grafo deve conter pelo menos uma estrutura $K_5^-$,
  o que exige que duas cópias de $K_5$ compartilhem pelo menos um triângulo.
  Como inicialmente temos apenas $H_0 \cong K_5$, devemos realizar pelo menos um passo de expansão
  degenerado do tipo $T_\ell$ para criar tal estrutura:
  \begin{itemize}
    \item Se realizamos dois ou mais passos $T_\ell$, o orçamento consumido por essas expansões é
          de pelo menos $2$. O orçamento restante para os passos $U_\ell$ é então no máximo $4 - 2 = 2$.
    \item Se realizamos exatamente um único passo $T_\ell$ (consumindo $1$ unidade de orçamento, com
          orçamento restante igual a $3$), analisamos a estrutura geométrica resultante.
  \end{itemize}
  Suponha, portanto, que realizamos exatamente um único passo $T_\ell$ sobre um triângulo
  $\Delta = \{a,b,c\} \subset H_0$, adicionando os vértices $x, y$ (com a aresta $xy$) conectados a $\Delta$.
  O conjunto de vértices de $H_0 \setminus \Delta$ é denotado por $A = \{u,v\}$, e os novos vértices
  são denotados por $Z = \{x,y\}$.

  As únicas não-arestas do grafo cujas adições podem fechar um $K_5^-$ sem criar imediatamente um $K_6$
  são os pares $\{w,z\}$ com $w \in A$ e $z \in Z$, cuja vizinhança comum contém o triângulo $\{a,b,c\}$.
  Essas não-arestas formam exatamente as arestas do grafo bipartido completo $A \times Z \cong K_{2,2}$.

  Teoricamente, o orçamento restante de $3$ permitiria realizar $3$ passos $U_\ell$ adicionando $3$ arestas
  deste $K_{2,2}$. No entanto, mostramos que a adição de qualquer par de arestas que compartilhe um vértice
  cria obrigatoriamente um $K_6$:
  \begin{enumerate}
    \item \emph{Compartilhamento em $Z$:} Suponha que adicionamos as arestas $xu$ e $xv$ (compartilhando $x \in Z$).
          Então, os seis vértices $\{a,b,c,x,u,v\}$ formam um $K_6$, pois $\{a,b,c\}$ é um triângulo,
          $u$ e $v$ pertencem a $H_0 \cong K_5$ (logo $uv$ é aresta e ambos se ligam a $\{a,b,c\}$),
          $x$ se liga a $\{a,b,c\}$ via $T_\ell$, e adicionamos as arestas $xu$ e $xv$. $\bot$
    \item \emph{Compartilhamento em $A$:} Suponha que adicionamos as arestas $xu$ e $yu$ (compartilhando $u \in A$).
          Então, os seis vértices $\{a,b,c,u,x,y\}$ formam um $K_6$, pois $\{a,b,c\}$ é um triângulo,
          $u$ se liga a $\{a,b,c\}$, $x$ e $y$ se ligam a $\{a,b,c\}$, $xy$ é a aresta de $T_\ell$,
          e adicionamos as arestas $xu$ e $yu$. $\bot$
  \end{enumerate}
  Portanto, para evitar a criação de um $K_6$, as arestas adicionadas pelos passos $U_\ell$ devem
  formar um acoplamento (matching) no grafo bipartido $A \times Z$. Como $|A| = 2$ e $|Z| = 2$,
  o tamanho máximo de um acoplamento em $K_{2,2}$ é exatamente $2$.

  Assim, mesmo com orçamento restante igual a $3$, o número máximo de passos $U_\ell$ realizáveis
  sem criar $K_6$ é no máximo $2$.

  \medskip
  \noindent\textit{Conclusão do Caso 1.}
  Reunindo os três subcasos, mostramos que se o bloco $B$ não contém $K_6$,
  qualquer caminho de construção indutiva limita o número total de passos $U_\ell$ a no máximo $\mathbf{2}$,
  o que conclui a prova da Afirmação~\ref{afirm:ulimit}.
\end{proof}

\section{Uma prova alternativa via colorações dos links à direita}

Nesta seção apresentamos uma prova determinística alternativa para a
coloração de grafos de densidade limitada, que evita $K_3$
monocromático, $K_4$ arco-íris e $K_5$ lexicográfico. Ao contrário da
abordagem por expansões (Definição de $R_\ell, T_\ell, Q_\ell, U_\ell$),
esta prova não depende da estrutura de $K_5$-bloco (fechamento,
função de penalidade): ela vale para \emph{qualquer} grafo finito com
densidade máxima $m(B) \leq 3$.

Ao longo de toda a seção, $B$ denota um grafo \textbf{finito}, o que
garante que todos os máximos e ínfimos considerados são atingidos.

\subsection{O teorema de densidade de Rödl--Ruciński}

Recordamos que, para um grafo $F$, definimos sua densidade máxima por
\[
  m(F)
  :=
  \max\left\{
  \frac{e(J)}{v(J)}
  :
  J\subseteq F,\ v(J)\geq 1
  \right\}.
\]

Para grafos $F$ e $H$ e um inteiro $r\geq 2$, escrevemos
\[
  F\xrightarrow{v}(H)_r
\]
quando toda $r$-coloração dos vértices de $F$ contém uma cópia
monocromática de $H$.

\begin{theorem}[Rödl--Ruciński {\cite[Theorem~5]{rodl1993lower}}]
  \label{thm:rr-vertex}
  Para todo grafo $H$ e todo inteiro $r\geq 2$, defina
  \[
    m_{\inf}^{(1)}(H,r)
    :=
    \inf\left\{
    m(F):
    F \text{ grafo finito},\ F\xrightarrow{v}(H)_r
    \right\}.
  \]
  Então
  \[
    \frac{1}{2}D(H)\,r
    \leq
    m_{\inf}^{(1)}(H,r),
    \qquad\text{onde } D(H):=\max_{J\subseteq H}\delta(J).
  \]
\end{theorem}

Aplicando o Teorema~\ref{thm:rr-vertex} ao grafo $H=K_4$ e a $r=2$,
temos $D(K_4)=\delta(K_4)=3$ (nenhum subgrafo próprio de $K_4$ tem
grau mínimo maior que $3$), logo
\[
  m_{\inf}^{(1)}(K_4,2)\geq \tfrac{1}{2}\cdot 3\cdot 2=3.
\]
Em particular,
\begin{equation}
  \label{eq:rr-k4}
  F\xrightarrow{v}(K_4)_2
  \quad\Longrightarrow\quad
  m(F)\geq 3.
\end{equation}

\subsection{Vértices bons e o lema estrutural}

Fixe um grafo finito $B$ munido de uma ordem total $\sigma$ sobre seus
vértices. Para cada $x\in V(B)$, definimos a \emph{vizinhança à
  direita} de $x$ por
\[
  N_\sigma^+(x):=\{y\in N_B(x): x<_\sigma y\}
\]
e o \emph{link à direita} de $x$ por $L_x:=B[N_\sigma^+(x)]$.

\begin{definition}[Vértice bom]
  \label{def:bom}
  Um vértice $x\in V(B)$ é \emph{bom} se $L_x\not\xrightarrow{v}(K_4)_2$,
  isto é, se existe uma $2$-coloração
  $\varphi_x:N_\sigma^+(x)\to\{0,1\}$ tal que nenhum $K_4$ de $L_x$ é
  monocromático sob $\varphi_x$.
\end{definition}

\begin{lemma}[Todos os vértices são bons]
  \label{lem:todos-bons}
  Seja $B$ um grafo finito e ordenado com $m(B)\leq 3$. Então todo
  $x\in V(B)$ é bom.
\end{lemma}

\begin{proof}
  Suponha, por contradição, que exista $x\in V(B)$ ruim, isto é,
  $L_x\xrightarrow{v}(K_4)_2$. Por \eqref{eq:rr-k4}, $m(L_x)\geq 3$.

  Como $L_x$ é finito, o máximo na definição de $m(L_x)$ é atingido:
  existe $J\subseteq L_x$ com $e(J)/v(J)=m(L_x)\geq 3$, ou seja,
  $e(J)\geq 3v(J)$. (Sem perda de generalidade $J$ pode ser tomado
  induzido, já que acrescentar arestas só aumenta a razão
  $e(J)/v(J)$.)

  Como $J$ é simples, $e(J)\leq\binom{v(J)}{2}$. Combinando,
  \[
    3v(J)\leq\frac{v(J)(v(J)-1)}{2}
    \quad\Longrightarrow\quad
    6\leq v(J)-1
    \quad\Longrightarrow\quad
    v(J)\geq 7.
  \]

  Como $V(J)\subseteq N_\sigma^+(x)$, o vértice $x$ é adjacente, em
  $B$, a todo vértice de $J$. Considere o \emph{cone} $C:=x+J$, com
  \[
    V(C):=V(J)\cup\{x\},\qquad
    E(C):=E(J)\cup\{xy: y\in V(J)\}.
  \]
  Como $x$ é adjacente a todos os vértices de $J$, temos $C\subseteq
    B$. Além disso, $v(C)=v(J)+1$ e $e(C)=e(J)+v(J)\geq 4v(J)$.

  Como $v(J)\geq 7>3$, temos $4v(J)>3v(J)+3=3(v(J)+1)$, ou seja,
  \[
    \frac{e(C)}{v(C)}\geq\frac{4v(J)}{v(J)+1}>3.
  \]
  Como $C\subseteq B$, isso implica $m(B)\geq e(C)/v(C)>3$,
  contradizendo a hipótese $m(B)\leq 3$.

  Portanto nenhum vértice ruim existe: todo $x\in V(B)$ é bom.
\end{proof}

\subsection{Construção da coloração}

Escreva $V(B)=\{v_1,\ldots,v_n\}$ com $v_1<_\sigma\cdots<_\sigma v_n$.
Pelo Lema~\ref{lem:todos-bons}, para cada $i\in[n]$ podemos escolher
\[
  \varphi_i:N_\sigma^+(v_i)\longrightarrow\{0,1\}
\]
tal que nenhum $K_4$ de $L_{v_i}$ seja monocromático sob $\varphi_i$.
Definimos $\psi:E(B)\to\{0,1\}$ por
\[
  \psi(v_iv_j):=\varphi_i(v_j),\qquad i<j.
\]

\begin{remark}[Ausência de conflito]
  Um mesmo vértice $v_j$ pode pertencer a vários conjuntos
  $N_\sigma^+(v_i)$, e os valores $\varphi_i(v_j)$ não precisam
  coincidir para diferentes $i$. Isso não gera incompatibilidade:
  toda aresta $v_iv_j$ ($i<j$) tem um único menor extremo, $v_i$, e é
  colorida exatamente uma vez, por $\varphi_i$.
\end{remark}

Fixamos também uma função \textbf{injetiva} $\lambda:V(B)\to\mathbb{N}$
e uma função injetiva $\iota:\mathbb{N}\times\{0,1\}\to\mathbb{N}$
(por exemplo $\iota(a,b)=2a+b$), e definimos a coloração final
\[
  \chi(v_iv_j):=\iota\bigl(\lambda(v_i),\,\psi(v_iv_j)\bigr)
  \;=\;\bigl(\lambda(v_i),\varphi_i(v_j)\bigr),
  \qquad i<j.
\]

\begin{theorem}[Coloração determinística do bloco]
  \label{thm:coloracao-bloco}
  Seja $B$ um grafo finito munido de ordem total $\sigma$, com
  $m(B)\leq 3$. A coloração $\chi$ definida acima não possui:
  \begin{enumerate}
    \item[(i)] $K_3$ monocromático;
    \item[(ii)] $K_4$ arco-íris;
    \item[(iii)] $K_5$ min-lex com respeito a $\sigma$;
    \item[(iv)] $K_5$ max-lex com respeito a $\sigma$.
  \end{enumerate}
\end{theorem}

\begin{proof}
  \textbf{(i) Ausência de $K_3$ monocromático.}
  Para $v_i<_\sigma v_j<_\sigma v_k$, temos
  $\chi(v_iv_j)=(\lambda(v_i),\varphi_i(v_j))$ e
  $\chi(v_jv_k)=(\lambda(v_j),\varphi_j(v_k))$. Como $\lambda$ é
  injetiva e $v_i\neq v_j$, $\lambda(v_i)\neq\lambda(v_j)$, logo
  $\chi(v_iv_j)\neq\chi(v_jv_k)$ independentemente das cores
  auxiliares. O triângulo não é monocromático.

  \medskip
  \noindent\textbf{(ii) Ausência de $K_4$ arco-íris.}
  Seja $v_i<_\sigma v_j<_\sigma v_k<_\sigma v_\ell$ uma cópia de
  $K_4$. As três arestas incidentes ao vértice mínimo $v_i$ têm todas
  primeira coordenada $\lambda(v_i)$; suas segundas coordenadas estão
  em $\{0,1\}$, logo, pelo princípio da casa dos pombos, duas delas
  coincidem. Portanto duas arestas do $K_4$ têm a mesma cor completa,
  e o $K_4$ não é arco-íris.

  \medskip
  \noindent\textbf{(iii) Ausência de $K_5$ min-lex.}
  Seja $H\cong K_5$ com vértices
  $v_{i_1}<_\sigma\cdots<_\sigma v_{i_5}$. Como $H$ é completo, os
  quatro vértices $v_{i_2},\ldots,v_{i_5}$ pertencem a
  $N_\sigma^+(v_{i_1})$ e induzem uma cópia de $K_4$ em
  $L_{v_{i_1}}$. Pela escolha de $\varphi_{i_1}$, os valores
  $\varphi_{i_1}(v_{i_2}),\ldots,\varphi_{i_1}(v_{i_5})$ não são todos
  iguais. Logo as quatro arestas incidentes a $v_{i_1}$ dentro de $H$
  não têm todas a mesma segunda coordenada, e portanto não têm todas
  a mesma cor $\chi$ — o que uma cópia min-lex exigiria. Assim $H$
  não é min-lex.

  \medskip
  \noindent\textbf{(iv) Ausência de $K_5$ max-lex.}
  Note que este item não utiliza a hipótese $m(B)\leq 3$ nem a
  escolha específica de $\varphi_i$: ele vale para \emph{qualquer}
  coloração da forma $\chi(v_iv_j)=(\lambda(v_i),\varphi_i(v_j))$ com
  $\lambda$ injetiva.
  Suponha, por contradição, que $H=\{x_1<_\sigma\cdots<_\sigma
    x_5\}\cong K_5$ seja max-lex sob $\chi$. As arestas
  $x_1x_3$ e $x_2x_3$ têm o mesmo maior extremo $x_3$, logo a
  condição max-lex exige $\chi(x_1x_3)=\chi(x_2x_3)$. Mas
  \[
    \chi(x_1x_3)=(\lambda(x_1),\varphi_{x_1}(x_3)),\qquad
    \chi(x_2x_3)=(\lambda(x_2),\varphi_{x_2}(x_3)),
  \]
  e como $\lambda$ é injetiva e $x_1\neq x_2$,
  $\lambda(x_1)\neq\lambda(x_2)$, logo
  $\chi(x_1x_3)\neq\chi(x_2x_3)$ — uma contradição.
  (O mesmo argumento vale para $K_t$ max-lex com qualquer $t\geq 3$.)
\end{proof}

\begin{remark}[Vértices monocromáticos locais não são um problema]
  A condição sobre $\varphi_x$ não exige cores distintas nem classes
  de cor pequenas: exige apenas que nenhuma classe contenha uma
  cópia de $K_4$. Uma classe de $\varphi_x$ pode conter vértices
  isolados, arestas, triângulos, ou grafos maiores livres de $K_4$ —
  a única configuração proibida é $y_1,y_2,y_3,y_4\in N_\sigma^+(x)$
  formando $K_4$ com $\varphi_x(y_1)=\cdots=\varphi_x(y_4)$, que
  criaria um $K_5=\{x,y_1,y_2,y_3,y_4\}$ min-lex.
\end{remark}

\begin{remark}[Relação com o procedimento de Rödl--Ruciński]
  O procedimento admite leitura sequencial: processamos
  $v_1,\ldots,v_n$ em ordem; ao processar $v_i$, colorimos as arestas
  ainda não coloridas $v_iv_j$ ($j>i$) segundo $\varphi_i$. Após
  processar $v_i$, nenhuma decisão futura pode criar um $K_5$
  min-lex com vértice mínimo $v_i$, pois suas quatro arestas dentro
  de qualquer tal $K_5$ já foram definitivamente coloridas de forma
  não constante na coordenada auxiliar. A diferença em relação ao
  procedimento clássico é trabalhar com $N_\sigma^+(v_i)$ em vez da
  vizinhança inteira — o que basta, pois um $K_5$ com vértice mínimo
  $v_i$ usa apenas vizinhos posteriores de $v_i$.
\end{remark}

\begin{remark}[Independência da estrutura de bloco]
  O Teorema~\ref{thm:coloracao-bloco} não usa em nenhum momento
  a noção de $K_5$-fechamento ou $K_5$-bloco: a única hipótese é
  $m(B)\leq 3$. Isso simplifica substancialmente a prova do análogo
  do Lema~\ref{lem:0-statement-K5}, que dependia da classificação das expansões
  em $R_\ell,T_\ell,Q_\ell,U_\ell$ e do controle do número de passos
  $U_\ell$. A decomposição em blocos permanece útil no algoritmo
  global — por exemplo, para permitir paletas de cores disjuntas
  entre blocos que compartilham vértices, já que a densidade de uma
  união de dois grafos de densidade $\leq 3$ que se intersectam em
  poucos vértices pode ultrapassar $3$ — mas deixa de ser necessária
  para a prova de que \emph{um único} bloco admite a coloração
  desejada.
\end{remark}
